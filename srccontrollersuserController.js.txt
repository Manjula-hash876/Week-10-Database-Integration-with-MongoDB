// src/controllers/userController.js

const jwt = require('jsonwebtoken');

const User = require('../models/User');

// ===============================
// Generate JWT Token
// ===============================

const generateAuthenticationToken = (userId) => {
  return jwt.sign(
    { userId },
    process.env.JWT_SECRET_KEY,
    {
      expiresIn: '7d'
    }
  );
};

// ===============================
// Register New User
// ===============================

const registerUser = async (request, response) => {
  try {
    const {
      name,
      email,
      password
    } = request.body;

    // Check if user already exists
    const existingUser = await User.findOne({ email });

    if (existingUser) {
      return response.status(400).json({
        success: false,
        message: 'User already exists with this email address'
      });
    }

    // Create new user account
    const newlyRegisteredUser = await User.create({
      name,
      email,
      password
    });

    // Generate JWT token
    const authenticationToken = generateAuthenticationToken(
      newlyRegisteredUser._id
    );

    response.status(201).json({
      success: true,
      message: 'User registered successfully',
      token: authenticationToken,
      data: {
        id: newlyRegisteredUser._id,
        name: newlyRegisteredUser.name,
        email: newlyRegisteredUser.email
      }
    });
  } catch (userRegistrationError) {
    response.status(500).json({
      success: false,
      message: 'Failed to register user',
      error: userRegistrationError.message
    });
  }
};

// ===============================
// Login User
// ===============================

const loginUser = async (request, response) => {
  try {
    const {
      email,
      password
    } = request.body;

    // Find user and include password field
    const existingUser = await User.findOne({ email }).select('+password');

    if (!existingUser) {
      return response.status(401).json({
        success: false,
        message: 'Invalid email or password'
      });
    }

    // Compare entered password
    const isPasswordValid = await existingUser.comparePassword(password);

    if (!isPasswordValid) {
      return response.status(401).json({
        success: false,
        message: 'Invalid email or password'
      });
    }

    // Generate JWT token
    const authenticationToken = generateAuthenticationToken(
      existingUser._id
    );

    response.status(200).json({
      success: true,
      message: 'Login successful',
      token: authenticationToken,
      data: {
        id: existingUser._id,
        name: existingUser.name,
        email: existingUser.email
      }
    });
  } catch (userLoginError) {
    response.status(500).json({
      success: false,
      message: 'Failed to login user',
      error: userLoginError.message
    });
  }
};

// ===============================
// Get Logged-In User Profile
// ===============================

const getUserProfile = async (request, response) => {
  try {
    const authenticatedUserId = request.userId;

    const authenticatedUser = await User.findById(
      authenticatedUserId
    );

    if (!authenticatedUser) {
      return response.status(404).json({
        success: false,
        message: 'User profile not found'
      });
    }

    response.status(200).json({
      success: true,
      data: {
        id: authenticatedUser._id,
        name: authenticatedUser.name,
        email: authenticatedUser.email,
        createdAt: authenticatedUser.createdAt
      }
    });
  } catch (userProfileError) {
    response.status(500).json({
      success: false,
      message: 'Failed to retrieve user profile',
      error: userProfileError.message
    });
  }
};

// ===============================
// Export Controller Functions
// ===============================

module.exports = {
  registerUser,
  loginUser,
  getUserProfile
};
